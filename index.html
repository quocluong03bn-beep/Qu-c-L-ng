} function getData() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Data'); var data = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < data.length; i++) { if (data[i][0] !== "") { result.push({ name: data[i][0], imei: data[i][1], condition: data[i][2], price: data[i][3], status: data[i][4] || 'Còn máy', image: data[i][5] || '', id: data[i][6] }); } } return result; // Sắp xếp hiển thị mới nhất lên đầu sẽ xử lý ở giao diện } function saveImage(base64Data, fileName) { try { var splitBase = base64Data.split(','); var type = splitBase[0].split(';')[0].replace('data:', ''); var byteCharacters = Utilities.base64Decode(splitBase[1]); var blob = Utilities.newBlob(byteCharacters, type, fileName); var folder; var folderIterator = DriveApp.getFoldersByName("Anh_Kho_May"); if (folderIterator.hasNext()) { folder = folderIterator.next(); } else { folder = DriveApp.createFolder("Anh_Kho_May"); folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } var file = folder.createFile(blob); return "https://drive.google.com/uc?export=view&id=" + file.getId(); } catch (e) { return ""; } } function addPhone(phone) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Data'); var id = new Date().getTime().toString(); var imageUrl = phone.image || ""; if (phone.imageBase64) { imageUrl = saveImage(phone.imageBase64, "IMG_" + phone.imei + "_" + id); } sheet.appendRow([phone.name, phone.imei, phone.condition, phone.price, phone.status, imageUrl, id]); return getData(); } function editPhone(id, phone) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Data'); var data = sheet.getDataRange().getValues(); for(var i = 1; i < data.length; i++){ if(data[i][6] == id){ // Cột G là index 6 var imageUrl = phone.image; if (phone.imageBase64) { imageUrl = saveImage(phone.imageBase64, "IMG_" + phone.imei + "_" + id); } sheet.getRange(i+1, 1, 1, 6).setValues([[phone.name, phone.imei, phone.condition, phone.price, phone.status, imageUrl]]); break; } } return getData(); } function deletePhone(id) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('Data'); var data = sheet.getDataRange().getValues(); for(var i = 1; i < data.length; i++){ if(data[i][6] == id){ sheet.deleteRow(i+1); break; } } return getData(); }
